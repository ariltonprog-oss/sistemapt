        });
    }
});

// 2. Função de Cálculo (Única)
function calcularVencimentoDireto(idRealizacao, idValidade, meses) {
    const inputR = document.getElementById(idRealizacao);
    const inputV = document.getElementById(idValidade);
    if (!inputR || !inputR.value || !inputV) return;

    const d = new Date(inputR.value + 'T00:00:00');
    d.setMonth(d.getMonth() + parseInt(meses));
    inputV.value = d.toISOString().split('T')[0];
}

// 3. Atualização da Grade (Visual)
function atualizarMatrizNRs() {
    const funcao = document.getElementById('funcao').value;
    const container = document.getElementById('containerMatrizNRs');
    if (!container) return;

    if (!funcao || !matrizNRsPorFuncao[funcao]) {
        container.innerHTML = '<p>Selecione uma função para ver as NRs.</p>';
        return;
    }

    let html = '<div class="grid-nrs">';
    matrizNRsPorFuncao[funcao].forEach(nr => {
        const d = bibliotecaRequisitos[nr];
        html += `<div class="card-nr-item">
            <label><strong>${nr}</strong></label>
            <small>${d.nome}</small>
            
            <!-- Campo de Realização -->
            <div class="form-group">
                <input type="date" id="realizacao_${nr}" onchange="calcularVencimentoDireto('realizacao_${nr}', 'validade_${nr}', ${d.validade})">
            </div>
            
            <!-- Campo de Validade (Agora com as classes para ocultar) -->
            <div class="form-group bloco-validade campo-oculto">
                <input type="date" id="validade_${nr}" readonly>
            </div>
        </div>`;
    });
    container.innerHTML = html + '</div>';

    // Chama a função de configuração de tela logo após criar os elementos.
    // Isso garante que, se estivermos no modo "Edição", os campos que 
    // acabaram de nascer ocultos sejam imediatamente revelados.
    if (typeof configurarModoDaTela === 'function') {
        configurarModoDaTela();
    }
}

// 4. Envio dos Dados (Compatível com Java)
async function processarEnvioFormulario() {
    const form = document.getElementById('formColaborador');
    if (!form.checkValidity()) return form.reportValidity();
    const usuarioLogado = JSON.parse(sessionStorage.getItem('usuarioLogado'));

    const data = {
        // Dados básicos
        nome: document.getElementById('nome').value.trim(),
        matricula: document.getElementById('matricula').value.trim(),
        funcao: document.getElementById('funcao').value,
        perfil: document.getElementById('perfil').value,

        // Login
        usuario: document.getElementById('usuario').value.trim(),
        senha: document.getElementById('senha').value,

        // Empresa
        ehTerceiro: document.getElementById('ehTerceiro').value === "true",

        // Empresa vinculada ao usuário logado
        empresa: usuarioLogado ? {
            id: usuarioLogado.empresaId
        } : null,

        // Permissões
        podeEmitirPt: document.getElementById('podeEmitirPT')?.checked || false,

        podeSolicitarPt:
            document.getElementById('perfil').value === "SOLICITANTE",

        // ASO
        realizacaoASO: document.getElementById('realizacaoASO').value || null,
        validadeASO: document.getElementById('validadeASO').value || null,

        // Reciclagem PT
        realizacaoReciclagemPt: document.getElementById('realizacaoReciclagemPt').value || null,
        validadeReciclagemPt: document.getElementById('validadeReciclagemPt').value || null,

        // Usuário que está realizando o cadastro
        usuarioLogadoId: usuarioLogado ? usuarioLogado.id : null


    };

    // 2. Loop Dinâmico corrigido para o padrão da sua classe
    document.querySelectorAll('.card-nr-item').forEach(card => {

        const nr = card.querySelector("strong").innerText.trim();

        const nomeCampo = obterNomeCampoJava(nr);

        if (!nomeCampo)
            return;

        const realizacao = card.querySelector('input[id^="realizacao_"]');

        const validade = card.querySelector('input[id^="validade_"]');

        if (!realizacao || !validade)
            return;

        if (realizacao.value) {

            data["realizacao" + nomeCampo] = realizacao.value;

            data["validade" + nomeCampo] = validade.value;

        }

    });
    console.log("Enviando funcionário:");
    console.log(JSON.stringify(data, null, 2));
    const url = "http://localhost:8080/api/funcionarios";

    try {
        const resp = await fetch(url, {
            method: "POST",
            headers: {
                "Content-Type": "application/json"
            },
            body: JSON.stringify(data)
        });

        if (!resp.ok) {

            const erro = await resp.text();

            console.error("Erro do backend:");
            console.error(erro);

            throw new Error(erro);

        }

        const funcionarioSalvo = await resp.json();

        console.log("Funcionário salvo:");
        console.log(funcionarioSalvo);

        alert("Funcionário cadastrado com sucesso!");
    } catch (erro) {
        console.error("Erro ao processar envio:", erro);
        alert("Erro ao cadastrar funcionário. Verifique o console para mais detalhes.");
    }
}

function toggleEmpresa() {

    const grupoEmpresa = document.getElementById('grupoEmpresa');
    if (grupoEmpresa) {
        grupoEmpresa.style.display = ehTerceiro ? 'block' : 'none';
    }
}

// Função atualizada e completa no cadastro.js
async function configurarModoDaTela() {
    const urlParams = new URLSearchParams(window.location.search);
    const idFuncionario = urlParams.get('id');

    // Se NÃO tiver ID na URL, é um Cadastro Novo. Não fazemos nada.
    if (!idFuncionario) return; 

    // MODO EDIÇÃO: Revelar os campos de validade
    const blocosValidade = document.querySelectorAll('.bloco-validade');
    blocosValidade.forEach(bloco => {
        bloco.classList.remove('campo-oculto');
    });

    // === INÍCIO DO CARREGAMENTO DE DADOS ===
    try {
        // 1. Busca os dados do colaborador específico no Java
        const response = await fetch(`http://localhost:8080/api/funcionarios/${idFuncionario}`);
        
        if (!response.ok) {
            throw new Error(`Erro ao buscar colaborador: ${response.status}`);
        }
        
        const funcionario = await response.json();
        console.log("Dados recebidos para edição:", funcionario);

        // 2. Preenche os dados básicos do formulário
        document.getElementById('nome').value = funcionario.nome || '';
        document.getElementById('matricula').value = funcionario.matricula || '';
        document.getElementById('usuario').value = funcionario.usuario || '';
        
        // Cuidado com a senha: em telas de edição, geralmente não retornamos a senha do back-end por segurança.
        // O ideal é deixar o campo em branco e só atualizar se o usuário digitar uma nova.
        
        document.getElementById('perfil').value = funcionario.perfil || '';
        document.getElementById('ehTerceiro').value = funcionario.ehTerceiro ? "true" : "false";
        toggleEmpresa(); // Atualiza a visibilidade do campo empresa

        if (document.getElementById('podeEmitirPT')) {
            document.getElementById('podeEmitirPT').checked = funcionario.podeEmitirPt || false;
        }

        // 3. Preenche as chaves FIXAS (Usando os nomes corrigidos do Java)
        if (document.getElementById('realizacaoASO')) {
            document.getElementById('realizacaoASO').value = funcionario.realizacaoASO || '';
        }
        if (document.getElementById('validadeASO')) {
            document.getElementById('validadeASO').value = funcionario.validadeASO || '';
        }
        
        if (document.getElementById('realizacaoReciclagemPt')) {
            document.getElementById('realizacaoReciclagemPt').value = funcionario.realizacaoReciclagemPt || '';
        }
        if (document.getElementById('validadeReciclagemPt')) {
            document.getElementById('validadeReciclagemPt').value = funcionario.validadeReciclagemPt || '';
        }

        // 4. A Mágica das NRs (Ordem correta!)
        // Primeiro injetamos a função
        const selectFuncao = document.getElementById('funcao');
        selectFuncao.value = funcionario.funcao || '';
        
        // Em seguida, disparamos a criação visual das caixinhas das NRs na tela
        atualizarMatrizNRs(); 

        // Por fim, injetamos as datas dentro das caixinhas recém-criadas
        if (funcionario.funcao && matrizNRsPorFuncao[funcionario.funcao]) {
            matrizNRsPorFuncao[funcionario.funcao].forEach(nr => {
                const nomeCampoJava = obterNomeCampoJava(nr);
                
                if (nomeCampoJava) {
                    const idRealizacao = `realizacao_${nr}`;
                    const idValidade = `validade_${nr}`;
                    
                    const inputR = document.getElementById(idRealizacao);
                    const inputV = document.getElementById(idValidade);
                    
                    if (inputR) inputR.value = funcionario["realizacao" + nomeCampoJava] || '';
                    if (inputV) inputV.value = funcionario["validade" + nomeCampoJava] || '';
                }
            });
        }

    } catch (error) {
        console.error("Erro ao carregar dados para edição:", error);
        alert("Erro ao tentar carregar os dados do colaborador para edição.");
    }
}

    if (idFuncionario) {
        // MODO EDIÇÃO: Remove a classe que esconde, fazendo os campos aparecerem
        const blocosValidade = document.querySelectorAll('.bloco-validade');
        blocosValidade.forEach(bloco => {
            bloco.classList.remove('campo-oculto');

        });
        isInitializing = false; // <--- LIBERA A TRAVA
    }
