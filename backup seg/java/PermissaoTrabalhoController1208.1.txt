package com.gestaopt.sistemapt.controller;

import java.time.LocalDate;
import java.util.ArrayList;
import java.util.List;

import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.CrossOrigin;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PatchMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.PutMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RestController;

import com.gestaopt.sistemapt.dto.BaixaPtRequestDTO;
import com.gestaopt.sistemapt.dto.EmitirPermissaoRequestDTO;
import com.gestaopt.sistemapt.model.Especialidade;
import com.gestaopt.sistemapt.model.Funcionario;
import com.gestaopt.sistemapt.model.PermissaoTrabalho;
import com.gestaopt.sistemapt.repository.FuncionarioRepository;
import com.gestaopt.sistemapt.repository.PermissaoTrabalhoRepository;
import com.gestaopt.sistemapt.service.PermissaoTrabalhoService;

@RestController
@RequestMapping("/api/permissoes-trabalho")
@CrossOrigin(origins = {"http://127.0.0.1:5500", "http://localhost:5500"})
public class PermissaoTrabalhoController {

    private final PermissaoTrabalhoRepository repository;
    private final FuncionarioRepository funcionarioRepository;
    private final PermissaoTrabalhoService service;

    public PermissaoTrabalhoController(PermissaoTrabalhoRepository repository,
            FuncionarioRepository funcionarioRepository,
            PermissaoTrabalhoService service) {
        this.repository = repository;
        this.funcionarioRepository = funcionarioRepository;
        this.service = service;
    }

    // 1. LISTAR TODAS AS PTs
    @GetMapping
    public List<PermissaoTrabalho> listarTodas() {
        return repository.findAll();
    }

    private PermissaoTrabalho ultimaPtRegistrada;
    private long instanteUltimoRegistro = 0;

    // 2. SALVAR NOVA PT COM VALIDAÇÃO DEFENSIVA
    @PostMapping
    public ResponseEntity<?> criar(@RequestBody PermissaoTrabalho pt) {
        if (pt == null) {
            return ResponseEntity.badRequest().body("❌ Erro: O corpo da requisição está vazio.");
        }

        long instanteAtual = System.currentTimeMillis();

        // 🛡️ TRAVA ANTI-DUPLICIDADE
        if (ultimaPtRegistrada != null && (instanteAtual - instanteUltimoRegistro < 5000)) {
            if (pt.getPlantaArea() != null && pt.getPlantaArea().equals(ultimaPtRegistrada.getPlantaArea())
                    && pt.getTag() != null && pt.getTag().equals(ultimaPtRegistrada.getTag())
                    && pt.getOrdemPj() != null && pt.getOrdemPj().equals(ultimaPtRegistrada.getOrdemPj())) {

                return ResponseEntity.status(409).body("Solicitação em duplicidade bloqueada pelo servidor.");
            }
        }

        // 🛡️ TRAVA: EXCLUSÃO MÚTUA DA MATRIZ DE RISCO (QUENTE vs FRIO)
        if (pt.isRequerTrabalhoQuente() && pt.isRequerTrabalhoFrio()) {
            return ResponseEntity.badRequest().body("🚫 BLOQUEIO SESMT: A atividade não pode ser classificada como Trabalho a Quente e Trabalho a Frio simultaneamente.");
        }

        // 🛡️ TRAVA: AST OBRIGATÓRIA NA SOLICITAÇÃO
        if (pt.getNomeAst() == null || pt.getNomeAst().isBlank()) {
            return ResponseEntity.badRequest().body("🚫 BLOQUEIO SESMT: O preenchimento da AST é obrigatório no momento da solicitação.");
        }

        // 👮‍♂️ FILTRO DO SOLICITANTE
        if (pt.getSolicitante() == null || pt.getSolicitante().getId() == null) {
            return ResponseEntity.badRequest().body("Erro: Solicitante não especificado na requisição.");
        }

        Long solicitanteId = java.util.Objects.requireNonNull(pt.getSolicitante().getId());
        Funcionario solicitante = funcionarioRepository.findById(solicitanteId).orElse(null);

        pt.setSolicitante(solicitante);

        LocalDate hoje = LocalDate.now();

        // Regra 1: ASO do Solicitante Vencido
        if (solicitante.getValidadeASO() != null && solicitante.getValidadeASO().isBefore(hoje)) {
            return ResponseEntity.status(403).body("🚫 SOLICITAÇÃO BARRADA: O ASO do solicitante (" + solicitante.getNome() + ") está vencido.");
        }

        // Regra 2: Reciclagem de PT do Solicitante Vencida
        if (solicitante.getValidadeReciclagemPt() != null && solicitante.getValidadeReciclagemPt().isBefore(hoje)) {
            return ResponseEntity.status(403).body("🚫 SOLICITAÇÃO BARRADA: A Reciclagem de PT do solicitante está vencida.");
        }

        // Regra 3: Se for Área Classificada, exige data da NR-20 válida
        if (pt.isRequerAreaClassificada()) {
            if (solicitante.getValidadeNr20() == null || solicitante.getValidadeNr20().isBefore(hoje)) {
                return ResponseEntity.status(403).body("🚫 SOLICITAÇÃO BARRADA: Trabalho em Área Classificada requer treinamento de NR-20 ativo para o solicitante.");
            }
        }

        if (pt.getStatus() == null || pt.getStatus().isBlank()) {
            pt.setStatus("SOLICITADA");
        }

        if (pt.getEmpresaGestora() == null || pt.getEmpresaGestora().isBlank()) {
            if (solicitante.getEmpresa() != null) {
                pt.setEmpresaGestora(solicitante.getEmpresa().toString());
            }
        }

        // 🪖 PROCESSAMENTO INTELIGENTE DA MATRIZ DE EPIs E ESPECIALIDADE
        try {
            Especialidade esp = pt.getAreaAtuacao();
            if (esp != null) {
                List<String> atividades = new ArrayList<>();
                if (pt.isRequerTrabalhoQuente()) {
                    atividades.add("QUENTE");
                }
                if (pt.isRequerTrabalhoAltura()) {
                    atividades.add("ALTURA");
                }
                if (pt.isRequerEspacoConfinado()) {
                    atividades.add("CONFINADO");
                }
                if (pt.isRequerAltaTensaoSep()) {
                    atividades.add("NR10_SEP");
                }
                if (pt.isRequerRiscoEletrico()) {
                    atividades.add("ELETRICO");
                }
                if (pt.isRequerAreaClassificada()) {
                    atividades.add("AREA_CLASSIFICADA");
                }
                if (pt.isRequerAtividadePintura()) {
                    atividades.add("PINTURA");
                }

                if (pt.getAgentesQuimicos() != null && pt.getAgentesQuimicos().isAtivo()) {
                    atividades.add("QUIMICO");
                }
                if (pt.getDiphoterine() != null && pt.getDiphoterine().isAtivo()) {
                    atividades.add("QUIMICO");
                }
                if (pt.getRuido() != null && pt.getRuido().isAtivo()) {
                    atividades.add("RUIDO");
                }

                List<String> episCalculados = esp.calcularEpisDinamicos(atividades);
                pt.setEpisObrigatorios(episCalculados);
            }
        } catch (Exception e) {
            System.err.println("Aviso: Não foi possível calcular os EPIs automaticamente: " + e.getMessage());
        }

        PermissaoTrabalho novaPt = repository.save(pt);

        this.ultimaPtRegistrada = novaPt;
        this.instanteUltimoRegistro = instanteAtual;

        return ResponseEntity.ok(novaPt);
    }

    // 3. ATUALIZAR STATUS PARA EMITIDA (DELEGANDO PARA O SERVICE)
    @PatchMapping("/{id}/emitir")
    public ResponseEntity<?> emitirPermissao(
            @PathVariable Long id,
            @RequestBody EmitirPermissaoRequestDTO dto) {

        if (id == null) {
            return ResponseEntity.badRequest().body("❌ O ID da Permissão de Trabalho não pode ser nulo.");
        }

        try {
            PermissaoTrabalho ptEmitida = service.emitirPt(id, dto);
            return ResponseEntity.ok(ptEmitida);
        } catch (Exception e) {
            return ResponseEntity.badRequest().body("❌ " + e.getMessage());
        }
    }

    // 4. BUSCAR PT POR ID COM GUARDA DE NULIDADE
    @GetMapping("/{id}")
    public ResponseEntity<PermissaoTrabalho> buscarPorId(@PathVariable Long id) {
        if (id == null) {
            return ResponseEntity.badRequest().build();
        }

        return repository.findById(id)
                .map(ResponseEntity::ok)
                .orElse(ResponseEntity.notFound().build());
    }

    // 5. ATUALIZAR DADOS DA PT SOLICITADA COM VALIDAÇÕES
    @PutMapping("/{id}")
    public ResponseEntity<?> atualizarPT(@PathVariable Long id, @RequestBody PermissaoTrabalho ptAtualizada) {
        if (id == null) {
            return ResponseEntity.badRequest().body("❌ O ID da Permissão de Trabalho não pode ser nulo.");
        }
        if (ptAtualizada == null) {
            return ResponseEntity.badRequest().body("❌ Os dados para atualização são obrigatórios.");
        }

        return repository.findById(id)
                .map(ptExistente -> {
                    if ("EMITIDA".equalsIgnoreCase(ptExistente.getStatus()) || "ENCERRADA".equalsIgnoreCase(ptExistente.getStatus())) {
                        return ResponseEntity.badRequest().body("🚫 Apenas PTs no status SOLICITADA podem ser editadas.");
                    }

                    ptExistente.setPlantaArea(ptAtualizada.getPlantaArea());
                    ptExistente.setTurnoGrupo(ptAtualizada.getTurnoGrupo());
                    ptExistente.setTag(ptAtualizada.getTag());
                    ptExistente.setOrdemPj(ptAtualizada.getOrdemPj());
                    ptExistente.setDescricaoAtividade(ptAtualizada.getDescricaoAtividade());
                    ptExistente.setAreaAtuacao(ptAtualizada.getAreaAtuacao());
                    ptExistente.setSolicitanteExecutante(ptAtualizada.isSolicitanteExecutante());

                    ptExistente.setRequerTrabalhoQuente(ptAtualizada.isRequerTrabalhoQuente());
                    ptExistente.setRequerTrabalhoFrio(ptAtualizada.isRequerTrabalhoFrio());
                    ptExistente.setRequerTrabalhoAltura(ptAtualizada.isRequerTrabalhoAltura());
                    ptExistente.setRequerEspacoConfinado(ptAtualizada.isRequerEspacoConfinado());
                    ptExistente.setRequerRiscoEletrico(ptAtualizada.isRequerRiscoEletrico());
                    ptExistente.setRequerAltaTensaoSep(ptAtualizada.isRequerAltaTensaoSep());
                    ptExistente.setRequerAreaClassificada(ptAtualizada.isRequerAreaClassificada());
                    ptExistente.setRequerSegurancaMaquinas(ptAtualizada.isRequerSegurancaMaquinas());
                    ptExistente.setRequerLoto(ptAtualizada.isRequerLoto());
                    ptExistente.setRequerAtividadePintura(ptAtualizada.isRequerAtividadePintura());

                    // ✅ CORREÇÃO: Atualizando também os objetos de agentes químicos, diphoterine e ruído
                    ptExistente.setAgentesQuimicos(ptAtualizada.getAgentesQuimicos());
                    ptExistente.setDiphoterine(ptAtualizada.getDiphoterine());
                    ptExistente.setRuido(ptAtualizada.getRuido());

                    if (ptAtualizada.getExecutantes() != null) {
                        ptExistente.setExecutantes(ptAtualizada.getExecutantes());
                    }

                    // 🪖 RECALCULO DOS EPIs DINÂMICOS COM BASE NAS FLAGS ATUALIZADAS
                    try {
                        Especialidade esp = ptExistente.getAreaAtuacao();
                        if (esp != null) {
                            List<String> atividades = new ArrayList<>();
                            if (ptExistente.isRequerTrabalhoQuente()) {
                                atividades.add("QUENTE");
                            }
                            if (ptExistente.isRequerTrabalhoAltura()) {
                                atividades.add("ALTURA");
                            }
                            if (ptExistente.isRequerEspacoConfinado()) {
                                atividades.add("CONFINADO");
                            }
                            if (ptExistente.isRequerAltaTensaoSep()) {
                                atividades.add("NR10_SEP");
                            }
                            if (ptExistente.isRequerRiscoEletrico()) {
                                atividades.add("ELETRICO");
                            }
                            if (ptExistente.isRequerAreaClassificada()) {
                                atividades.add("AREA_CLASSIFICADA");
                            }
                            if (ptExistente.isRequerAtividadePintura()) {
                                atividades.add("PINTURA");
                            }

                            // ✅ CORREÇÃO: Incluindo a checagem correta dos riscos químicos e físicos no recálculo
                            if (ptExistente.getAgentesQuimicos() != null && ptExistente.getAgentesQuimicos().isAtivo()) {
                                atividades.add("QUIMICO");
                            }
                            if (ptExistente.getDiphoterine() != null && ptExistente.getDiphoterine().isAtivo()) {
                                atividades.add("QUIMICO");
                            }
                            if (ptExistente.getRuido() != null && ptExistente.getRuido().isAtivo()) {
                                atividades.add("RUIDO");
                            }

                            ptExistente.setEpisObrigatorios(esp.calcularEpisDinamicos(atividades));
                        }
                    } catch (Exception e) {
                        System.err.println("Aviso: Não foi possível recalcular os EPIs na atualização: " + e.getMessage());
                    }

                    PermissaoTrabalho salva = repository.save(ptExistente);
                    return ResponseEntity.ok(salva);
                })
                .orElse(ResponseEntity.notFound().build());
    }

    // 6. BAIXA DA PT (UNIFICADA E AUDITADA PELO SERVICE)
    @PostMapping("/{id}/baixa")
    public ResponseEntity<String> realizarBaixa(@PathVariable Long id, @RequestBody BaixaPtRequestDTO dto) {
        try {
            String relatorioAuditoria = service.realizarBaixaPt(id, dto);
            return ResponseEntity.ok(relatorioAuditoria);
        } catch (Exception e) {
            return ResponseEntity.badRequest().body("Erro ao processar baixa: " + e.getMessage());
        }
    }
}
