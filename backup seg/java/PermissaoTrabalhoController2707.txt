package com.gestaopt.sistemapt.controller;

import java.time.LocalDate;
import java.util.List;

import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.PutMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.bind.annotation.RestController;

import com.gestaopt.sistemapt.model.Funcionario;
import com.gestaopt.sistemapt.model.PermissaoTrabalho;
import com.gestaopt.sistemapt.repository.FuncionarioRepository;
import com.gestaopt.sistemapt.repository.PermissaoTrabalhoRepository;
import com.gestaopt.sistemapt.service.PermissaoTrabalhoService;

@RestController
@RequestMapping("/api/permissoes-trabalho")

public class PermissaoTrabalhoController {

    private final PermissaoTrabalhoRepository repository;
    private final FuncionarioRepository funcionarioRepository;
    private final PermissaoTrabalhoService service; // <-- ADICIONADO ATRIBUTO DO SERVICE

    // Construtor atualizado via Injeção de Dependência
    public PermissaoTrabalhoController(PermissaoTrabalhoRepository repository,
            FuncionarioRepository funcionarioRepository,
            PermissaoTrabalhoService service) {
        this.repository = repository;
        this.funcionarioRepository = funcionarioRepository;
        this.service = service;
    }

    // 1. LISTAR TODAS AS PTs
    @GetMapping
    public List<PermissaoTrabalho> listarTodas() {
        return repository.findAll();
    }

    private PermissaoTrabalho ultimaPtRegistrada;
    private long instanteUltimoRegistro = 0;

    // 2. SALVAR NOVA PT
    @PostMapping
    public ResponseEntity<?> criar(@RequestBody PermissaoTrabalho pt) {
        long instanteAtual = System.currentTimeMillis();

        // 🛡️ TRAVA ANTI-DUPLICIDADE
        if (ultimaPtRegistrada != null && (instanteAtual - instanteUltimoRegistro < 5000)) {
            if (pt.getPlantaArea() != null && pt.getPlantaArea().equals(ultimaPtRegistrada.getPlantaArea())
                    && pt.getTag() != null && pt.getTag().equals(ultimaPtRegistrada.getTag())
                    && pt.getOrdemPj() != null && pt.getOrdemPj().equals(ultimaPtRegistrada.getOrdemPj())) {

                return ResponseEntity.status(409).body("Solicitação em duplicidade bloqueada pelo servidor.");
            }
        }

        // 🛡️ NOVA TRAVA: EXCLUSÃO MÚTUA DA MATRIZ DE RISCO (QUENTE vs FRIO)
        if (pt.isRequerTrabalhoQuente() && pt.isRequerTrabalhoFrio()) {
            return ResponseEntity.badRequest().body("🚫 BLOQUEIO SESMT: A atividade não pode ser classificada como Trabalho a Quente e Trabalho a Frio simultaneamente.");
        }

        // 👮‍♂️ FILTRO DO SOLICITANTE
        if (pt.getSolicitante() == null || pt.getSolicitante().getId() == null) {
            return ResponseEntity.badRequest().body("Erro: Solicitante não especificado na requisição.");
        }

        // Busca o solicitante completo no banco para validar regras de negócio
        Funcionario solicitante = funcionarioRepository.findById(pt.getSolicitante().getId()).orElse(null);

        if (solicitante == null) {
            return ResponseEntity.badRequest().body("Erro: Solicitante não localizado no banco de dados.");
        }

        // 📝 GARANTE QUE O OBJETO PT ESTÁ COM O SOLICITANTE COMPLETO E VALIDADO
        pt.setSolicitante(solicitante);

        LocalDate hoje = LocalDate.now();

        // Regra 1: ASO do Solicitante Vencido
        if (solicitante.getValidadeASO() != null && solicitante.getValidadeASO().isBefore(hoje)) {
            return ResponseEntity.status(403).body("🚫 SOLICITAÇÃO BARRADA: O ASO do solicitante (" + solicitante.getNome() + ") está vencido.");
        }

        // Regra 2: Reciclagem de PT do Solicitante Vencida
        if (solicitante.getValidadeReciclagemPt() != null && solicitante.getValidadeReciclagemPt().isBefore(hoje)) {
            return ResponseEntity.status(403).body("🚫 SOLICITAÇÃO BARRADA: A Reciclagem de PT do solicitante está vencida.");
        }

        // Regra 3: Se for Área Classificada, exige data da NR-20 válida
        if (pt.isRequerAreaClassificada()) {
            if (solicitante.getValidadeNr20() == null || solicitante.getValidadeNr20().isBefore(hoje)) {
                return ResponseEntity.status(403).body("🚫 SOLICITAÇÃO BARRADA: Trabalho em Área Classificada requer treinamento de NR-20 ativo para o solicitante.");
            }
        }

        if (pt.getStatus() == null) {
            pt.setStatus("RASCUNHO");
        }

        if (pt.getEmpresaGestora() == null || pt.getEmpresaGestora().isBlank()) {
            if (solicitante.getEmpresa() != null) {
                // Pega o nome da empresa associada ao cadastro do funcionário/solicitante
                pt.setEmpresaGestora(solicitante.getEmpresa().toString());
            }
        }

        PermissaoTrabalho novaPt = repository.save(pt);

        this.ultimaPtRegistrada = novaPt;
        this.instanteUltimoRegistro = instanteAtual;

        return ResponseEntity.ok(novaPt);
    }

    // 3. ATUALIZAR STATUS PARA EMITIDA (COM VINCULAÇÃO DE EMITENTE LOGADO)
    @PutMapping("/{id}/emitir")
    public ResponseEntity<?> emitirPermissao(@PathVariable Long id, @RequestParam(name = "emitenteId", required = false) Long emitenteId) {
        return repository.findById(id)
                .map(pt -> {
                    try {
                        // Se o ID do emitente foi enviado, busca e vincula à PT antes de validar
                        if (emitenteId != null) {
                            Funcionario emitenteObj = funcionarioRepository.findById(emitenteId)
                                    .orElseThrow(() -> new RuntimeException("Funcionário emitente não localizado."));
                            pt.setEmitente(emitenteObj);
                        } else if (pt.getEmitente() == null) {
                            return ResponseEntity.badRequest().body("❌ Erro SESMT: Nenhum emitente foi identificado para esta operação.");
                        }

                        // O Service agora receberá a PT com o objeto do emitente devidamente preenchido
                        PermissaoTrabalho ptEmitida = service.validarEEmitirPT(pt);
                        return ResponseEntity.ok(ptEmitida);
                    } catch (Exception e) {
                        return ResponseEntity.badRequest().body("❌ " + e.getMessage());
                    }
                })
                .orElse(ResponseEntity.notFound().build());
    }

    // 4. BAIXA DA PT COM RELATÓRIO DE COMPLIANCE
    @PutMapping("/{id}/baixar")
    public ResponseEntity<?> baixarPermissao(@PathVariable Long id, @RequestBody List<Long> executantesIds) {
        return repository.findById(id)
                .map(pt -> {
                    LocalDate hoje = LocalDate.now();
                    StringBuilder relatorio = new StringBuilder();
                    relatorio.append("=== AUDITORIA DE COMPLIANCE SESMT ===\n");
                    relatorio.append("PT #").append(pt.getId()).append(" | OM: ").append(pt.getOrdemPj()).append("\n");

                    boolean temDesvio = false;
                    List<Funcionario> equipeCampo = funcionarioRepository.findAllById(executantesIds);

                    for (Funcionario f : equipeCampo) {
                        // 1. Valida ASO de quem foi a campo
                        if (f.getValidadeASO() != null && f.getValidadeASO().isBefore(hoje)) {
                            relatorio.append("⚠️ DESVIO: Executante ").append(f.getNome()).append(" trabalhou com ASO VENCIDO.\n");
                            temDesvio = true;
                        }

                        // 2. Se a PT requer Altura, valida a data da NR-35
                        if (pt.isRequerTrabalhoAltura()) {
                            if (f.getValidadeNr35() == null || f.getValidadeNr35().isBefore(hoje)) {
                                relatorio.append("⚠️ DESVIO: Executante ").append(f.getNome()).append(" atuou em Altura sem treinamento de NR-35 ativo ou vencido.\n");
                                temDesvio = true;
                            }
                        }

                        // 3. Se a PT requer Espaço Confinado, valida a data da NR-33
                        if (pt.isRequerEspacoConfinado()) {
                            if (f.getValidadeNr33() == null || f.getValidadeNr33().isBefore(hoje)) {
                                relatorio.append("⚠️ DESVIO: Executante ").append(f.getNome()).append(" entrou em Espaço Confinado sem treinamento de NR-33 ativo ou vencido.\n");
                                temDesvio = true;
                            }
                        }
                    }

                    if (!temDesvio) {
                        relatorio.append("🟢 Operação finalizada em total conformidade legal.\n");
                    }

                    pt.setExecutantes(equipeCampo);
                    pt.setStatus("ENCERRADA");
                    repository.save(pt);

                    return ResponseEntity.ok(new RespostaCompliance(relatorio.toString(), temDesvio));
                })
                .orElse(ResponseEntity.notFound().build());
    }
}

class RespostaCompliance {

    public String relatorio;
    public boolean possuiDesvios;

    public RespostaCompliance(String r, boolean d) {
        this.relatorio = r;
        this.possuiDesvios = d;
    }
}
