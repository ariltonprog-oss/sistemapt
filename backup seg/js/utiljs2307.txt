// O script abaixo percorre sua biblioteca e configura tudo sozinho
Object.keys(bibliotecaRequisitos).forEach(key => {
    const idRealizacao = `realizacao${key}`;
    const idValidade = `validade${key}`;
    const meses = bibliotecaRequisitos[key].validade;

    const inputRealizacao = document.getElementById(idRealizacao);

    if (inputRealizacao) {
        inputRealizacao.addEventListener('change', () => {
            calcularVencimentoDireto(idRealizacao, idValidade, meses);
        });
    }
});

// 2. Função de Cálculo (Única)
function calcularVencimentoDireto(idRealizacao, idValidade, meses) {
    const inputR = document.getElementById(idRealizacao);
    const inputV = document.getElementById(idValidade);
    if (!inputR || !inputR.value || !inputV) return;

    const d = new Date(inputR.value + 'T00:00:00');
    d.setMonth(d.getMonth() + parseInt(meses));
    inputV.value = d.toISOString().split('T')[0];
}
