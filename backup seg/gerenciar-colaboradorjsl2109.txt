// Variável global para armazenar a lista completa de colaboradores
let listaColaboradoresGlobal = [];

document.addEventListener('DOMContentLoaded', () => {
    // 1. Verificação de Segurança (Sessão)
    const logado = sessionStorage.getItem('usuarioLogado');
    if (!logado) {
        window.location.href = 'login.html';
        return;
    }

    const usuario = JSON.parse(logado);
    const perfil = usuario.perfil ? usuario.perfil.toUpperCase() : '';
    console.log("Gerenciamento de Colaboradores carregado. Acesso como: ", perfil);

    // Carrega os dados da API
    carregarColaboradores();

    // 2. Configura os eventos de escuta para busca e filtros em tempo real
    const campoBusca = document.getElementById('campoBusca');
    const filtroStatus = document.getElementById('filtroStatus');
    const filtroTreinamento = document.getElementById('filtroTreinamento');

    if (campoBusca) {
        campoBusca.addEventListener('input', aplicarFiltros);
    }
    if (filtroStatus) {
        filtroStatus.addEventListener('change', aplicarFiltros);
    }
    if (filtroTreinamento) {
        filtroTreinamento.addEventListener('change', aplicarFiltros);
    }
});

// Função para buscar colaboradores no backend
async function carregarColaboradores() {
    try {
        const response = await fetch('http://localhost:8080/api/funcionarios');
        if (!response.ok) {
            throw new Error('Erro ao buscar colaboradores na API.');
        }

        listaColaboradoresGlobal = await response.json();
        
        // Aplica o filtro inicial (por padrão exibe os "ativos")
        aplicarFiltros();

    } catch (error) {
        console.error("Erro ao carregar colaboradores:", error);
        const corpoTabela = document.getElementById('tabelaCorpo');
        if (corpoTabela) {
            corpoTabela.innerHTML = `
                <tr>
                    <td colspan="8" style="text-align: center; color: #ff6b6b; padding: 20px;">
                        Erro ao conectar com o servidor para carregar os colaboradores.
                    </td>
                </tr>`;
        }
    }
}

// Função centralizadora de filtragem (Texto + Status + Treinamentos)
function aplicarFiltros() {
    const termoBusca = document.getElementById('campoBusca')?.value.toLowerCase().trim() || '';
    const statusSelecionado = document.getElementById('filtroStatus')?.value || 'ativo';
    const treinamentoSelecionado = document.getElementById('filtroTreinamento')?.value || 'todos';

    const hoje = new Date();
    hoje.setHours(0, 0, 0, 0);

    const filtrados = listaColaboradoresGlobal.filter(func => {
        // 1. Filtro de Texto (Nome ou Matrícula)
        const nome = (func.nome || '').toLowerCase();
        const matricula = (func.matricula || '').toLowerCase();
        const matchTexto = nome.includes(termoBusca) || matricula.includes(termoBusca);

        // 2. Filtro de Status na Empresa (Ativo / Inativo)
        // Assume-se que o atributo possa ser func.ativo (boolean) ou func.status (string 'ativo'/'inativo')
        let isAtivo = true;
        if (typeof func.ativo !== 'undefined') {
            isAtivo = Boolean(func.ativo);
        } else if (func.status) {
            isAtivo = func.status.toLowerCase() === 'ativo';
        }

        let matchStatus = true;
        if (statusSelecionado === 'ativo') matchStatus = isAtivo;
        if (statusSelecionado === 'inativo') matchStatus = !isAtivo;

        // 3. Verificação de Vencimentos (ASO, Reciclagem PT ou NRs)
        let temVencimento = false;

        if (func.validadeASO) {
            const dataAso = new Date(func.validadeASO + 'T00:00:00');
            if (dataAso < hoje) temVencimento = true;
        } else {
            temVencimento = true; // Considera ausente como pendência
        }

        if (func.validadeReciclagemPt) {
            const dataRec = new Date(func.validadeReciclagemPt + 'T00:00:00');
            if (dataRec < hoje) temVencimento = true;
        }

        const qtdNrs = Number(func.nrsVencidas || func.quantidadeNrsVencidas || 0);
        if (qtdNrs > 0) temVencimento = true;

        let matchTreinamento = true;
        if (treinamentoSelecionado === 'regular') matchTreinamento = !temVencimento;
        if (treinamentoSelecionado === 'vencidos') matchTreinamento = temVencimento;

        return matchTexto && matchStatus && matchTreinamento;
    });

    renderizarTabela(filtrados);
}

// Renderiza os dados filtrados na tabela HTML
function renderizarTabela(lista) {
    const corpoTabela = document.getElementById('tabelaCorpo');
    if (!corpoTabela) return;

    if (lista.length === 0) {
        corpoTabela.innerHTML = `
            <tr>
                <td colspan="8" style="text-align: center; padding: 20px; color: #666;">
                    Nenhum colaborador encontrado com os filtros selecionados.
                </td>
            </tr>`;
        return;
    }

    const hoje = new Date();
    hoje.setHours(0, 0, 0, 0);

    corpoTabela.innerHTML = lista.map(func => {
        // Formatação e verificação de vencimento (ASO)
        let asoFormatado = '-';
        let classeAso = '';
        if (func.validadeASO) {
            const dataAso = new Date(func.validadeASO + 'T00:00:00');
            asoFormatado = dataAso.toLocaleDateString('pt-BR');
            if (dataAso < hoje) {
                classeAso = 'color: #ff6b6b; font-weight: bold;';
                asoFormatado += ' ⚠️ (Vencido)';
            }
        } else {
            asoFormatado = 'Ausente';
            classeAso = 'color: #ff9f43;';
        }

        // Formatação e verificação de validade de PT / Reciclagem
        let validadePtFormatado = '-';
        let classeValidadePt = '';
        if (func.validadeReciclagemPt) {
            const dataRec = new Date(func.validadeReciclagemPt + 'T00:00:00');
            validadePtFormatado = dataRec.toLocaleDateString('pt-BR');
            if (dataRec < hoje) {
                classeValidadePt = 'color: #ff6b6b; font-weight: bold;';
                validadePtFormatado += ' ⚠️ (Vencida)';
            }
        } else {
            validadePtFormatado = 'Ausente';
            classeValidadePt = 'color: #ff9f43;';
        }

        // NRs / Treinamentos
        const nrsVencidas = func.nrsVencidas || func.quantidadeNrsVencidas || 'Nenhuma';

        // Tratamento seguro para extrair o nome fantasia da empresa associada
        let nomeEmpresa = '-';
        if (func.empresa) {
            if (typeof func.empresa === 'object') {
                nomeEmpresa = func.empresa.nomeFantasia || func.empresa.nome || func.empresa.razaoSocial || '-';
            } else {
                nomeEmpresa = func.empresa;
            }
        }

        return `
            <tr>
                <td><strong>${func.nome || '-'}</strong></td>
                <td>${func.matricula || '-'}</td>
                <td>${func.funcao || '-'}</td>
                <td>${nomeEmpresa}</td>
                <td><span style="${classeAso}">${asoFormatado}</span></td>
                <td><span style="${classeValidadePt}">${validadePtFormatado}</span></td>
                <td>${nrsVencidas}</td>
                <td class="coluna-acoes">
                    <a href="editar-colaborador.html?id=${func.id}" class="btn-acao-editar" title="Editar Colaborador">✏️ Editar</a>
                </td>
            </tr>
        `;
    }).join('');
}
