let ptAtual = null;
let executantesReais = [];

document.addEventListener("DOMContentLoaded", function () {
    console.log("Gerenciar-PT inicializado com Delegação de Eventos.");

    // 1. Carregar dados da PT via URL
    async function carregarDadosPT() {
        const urlParams = new URLSearchParams(window.location.search);
        const ptId = urlParams.get('id');

        if (!ptId) {
            console.warn("Nenhum ID de PT informado na URL.");
            return;
        }

        try {
            const response = await fetch(`http://localhost:8080/api/pts/${ptId}`, {
                headers: {
                    'Authorization': 'Bearer ' + localStorage.getItem('token')
                }
            });

            if (!response.ok) {
                throw new Error(`Erro ao buscar dados da PT: ${response.status}`);
            }

            ptAtual = await response.json();

            if (document.getElementById('plantaArea')) document.getElementById('plantaArea').value = ptAtual.plantaArea || '';
            if (document.getElementById('turnoGrupo')) document.getElementById('turnoGrupo').value = ptAtual.turnoGrupo || '';
            if (document.getElementById('tag')) document.getElementById('tag').value = ptAtual.tag || '';
            if (document.getElementById('ordemPj')) document.getElementById('ordemPj').value = ptAtual.ordemPj || '';
            if (document.getElementById('descricaoAtividade')) document.getElementById('descricaoAtividade').value = ptAtual.descricaoAtividade || '';

        } catch (error) {
            console.error("Erro ao carregar a PT:", error);
        }
    }

    carregarDadosPT();

   // 2. Abertura do Modal com controle total de CSS e Classes
    document.addEventListener('click', function (e) {
        const btnRevalidar = e.target.closest('#btnAbrirModalReval');
        
        if (btnRevalidar) {
            e.preventDefault();
            console.log("Botão de revalidação acionado!");

            const modalReval = document.getElementById('modalRevalidacao');
            if (!modalReval) {
                alert("Erro crítico: O elemento #modalRevalidacao não foi encontrado no HTML.");
                return;
            }

            // Preenche os campos
            const solicitanteNome = (ptAtual && ptAtual.solicitante) ? ptAtual.solicitante : "Não informado";
            const campoSolicitante = document.getElementById('modalSolicitante');
            if (campoSolicitante) campoSolicitante.value = solicitanteNome;
            
            const agora = new Date();
            const campoData = document.getElementById('revalData');
            const campoHora = document.getElementById('revalHora');

            if (campoData) campoData.value = agora.toISOString().split('T')[0];
            if (campoHora) campoHora.value = agora.toTimeString().slice(0, 5);

            // FORÇA A EXIBIÇÃO: Remove o style antigo e aplica flex com prioridade máxima
            modalReval.removeAttribute('style');
            modalReval.style.cssText = "display: flex !important; visibility: visible !important; opacity: 1 !important; z-index: 99999 !important;";
            
            // Caso o seu CSS use classes para abrir (ex: .ativo ou .open), adicionamos também:
            modalReval.classList.add('ativo', 'open', 'show');
        }

        // Fechar o modal
        const btnFechar = e.target.closest('#btnFecharModal');
        if (btnFechar) {
            const modalReval = document.getElementById('modalRevalidacao');
            if (modalReval) {
                modalReval.style.cssText = "display: none !important;";
                modalReval.classList.remove('ativo', 'open', 'show');
            }
        }
    });})
