            <input type="text" id="usuario" required>
        </div>

            <input type="password" id="senha" required>

        <div id="mensagemErro" class="error"></div>

<script>
    const API_URL = "http://localhost:8080/api/auth/login"; // Ajuste para o seu endpoint de login

    document.getElementById('loginForm').addEventListener('submit', async (e) => {
        e.preventDefault();
        
        const usuario = document.getElementById('usuario').value;
        const senha = document.getElementById('senha').value;
        const msgErro = document.getElementById('mensagemErro');
        msgErro.innerText = "";

        try {
            const response = await fetch(API_URL, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ usuario, senha })
            });

            if (!response.ok) {
                throw new Error("Usuário ou senha inválidos.");
            }

            const dadosExemplo = await response.json();
            
            // Exemplo de lógica de direcionamento baseada no perfil (ROLE)
            localStorage.setItem('usuarioLogado', JSON.stringify(dadosExemplo));
            
            if (dadosExemplo.perfil === 'MASTER') {
                window.location.href = 'admin.html'; // Tela do Admin Master para criar empresas
            } else {
                window.location.href = 'dashboard.html'; // Dashboard geral
            }

        } catch (error) {
            msgErro.innerText = error.message;
        }
    });
</script>
