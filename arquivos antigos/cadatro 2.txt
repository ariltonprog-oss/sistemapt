<script>
    const API_URL = "http://localhost:8080/api/funcionarios";

    document.getElementById('funcionarioForm').addEventListener('submit', async (e) => {
        e.preventDefault();

        // Pegamos os dados salvos no login para saber qual empresa o Administrador gerencia
        const usuarioLogado = JSON.parse(localStorage.getItem('usuarioLogado')) || { empresaId: 1 };

        const colaboradorData = {
            nome: document.getElementById('nome').value,
            perfil: document.getElementById('perfil').value,
            ehTerceiro: document.getElementById('ehTerceiro').checked, // Envia true ou false puro
            empresaId: usuarioLogado.empresaId // Vincula automaticamente à empresa do Admin logado
        };

        try {
            const response = await fetch(API_URL, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(colaboradorData)
            });

            if (response.ok) {
                alert("Funcionário cadastrado com sucesso!");
                document.getElementById('funcionarioForm').reset();
            } else {
                const erro = await response.json();
                alert("Erro ao cadastrar: " + (erro.message || "Erro interno"));
            }
        } catch (error) {
            alert("Erro de conexão com o servidor Java.");
        }
    });
</script>
