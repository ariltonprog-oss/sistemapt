        // Objeto global de cache para dados vindos da edição
        let dadosTreinamentosDoBanco = {};

        // ==========================================
        // 1. DEFINIÇÃO DA MATRIZ E REQUISITOS (BIBLIOTECA)
        // ==========================================
        const matrizNRsPorFuncao = {
            "Operador Industrial": ["NR10", "NR11", "NR12", "NR33", "NR35"],
            "Mecânico": ["NR11", "NR12", "NR33", "NR35"],
            "Eletricista": ["NR10", "NR12", "NR35"],
            "Instrumentista": ["NR10", "NR12"],
            "Pintor Industrial": ["NR33", "NR35"],
            "Caldeireiro": ["NR12", "NR34", "NR35"],
            "Encanador Industrial": ["NR12", "NR33", "NR35"],
            "Soldador": ["NR12", "NR34", "NR35"]
        };

        const bibliotecaRequisitos = {
            "NR10": { nome: "NR-10 Segurança em Eletricidade", validade: 24 },
            "NR11": { nome: "NR-11 Transporte e Movimentação", validade: 12 },
            "NR12": { nome: "NR-12 Segurança em Máquinas", validade: 24 },
            "NR33": { nome: "NR-33 Espaço Confinado", validade: 12 },
            "NR34": { nome: "NR-34 Condições de Trabalho Naval/Quente", validade: 12 },
            "NR35": { nome: "NR-35 Trabalho em Altura", validade: 24 }
        };

        // ==========================================
        // 2. FUNÇÕES DE SUPORTE E CÁLCULOS DE VALIDADE
        // ==========================================
        function calcularVencimentoDireto(idRealizacao, idValidade, mesesValidade) {
            const inputRealizacao = document.getElementById(idRealizacao);
            const inputValidade = document.getElementById(idValidade);
            
            if (!inputRealizacao || !inputRealizacao.value || !inputValidade) return;
            if (!mesesValidade || parseInt(mesesValidade) <= 0) return;

            const dataRealizacao = new Date(inputRealizacao.value + 'T00:00:00');
            if (isNaN(dataRealizacao.getTime())) return;

            dataRealizacao.setMonth(dataRealizacao.getMonth() + parseInt(mesesValidade));
            
            const ano = dataRealizacao.getFullYear();
            const mes = String(dataRealizacao.getMonth() + 1).padStart(2, '0');
            const dia = String(dataRealizacao.getDate()).padStart(2, '0');

            inputValidade.value = `${ano}-${mes}-${dia}`;
        }

        function ajustarCamposLogin() {
            const perfil = document.getElementById('perfil').value;
            const usuarioInput = document.getElementById('usuario');
            const senhaInput = document.getElementById('senha');
            const containerAlcada = document.getElementById('containerAlcadaEmissao');

            // Controla a exibição do checkbox condicional para Encarregado
            if (perfil === 'ENCARREGADO') {
                if (containerAlcada) containerAlcada.classList.remove('elemento-escondido');
            } else {
                if (containerAlcada) containerAlcada.classList.add('elemento-escondido');
                const chkEmitir = document.getElementById('podeEmitirPT');
                if (chkEmitir) chkEmitir.checked = false;
            }

            if (perfil === 'EXECUTANTE' || perfil === '') {
                usuarioInput.value = '';
                senhaInput.value = '';
                usuarioInput.disabled = true;
                senhaInput.disabled = true;
                senhaInput.required = false;
            } else {
                usuarioInput.disabled = false;
                senhaInput.disabled = false;
                senhaInput.required = !document.getElementById('matricula').disabled;
            }
        }

        function toggleEmpresa() {
            const ehTerceiro = document.getElementById('ehTerceiro').value === 'true';
            const grupoEmpresa = document.getElementById('grupoEmpresa');
            const inputEmpresa = document.getElementById('empresa');

            if (grupoEmpresa && inputEmpresa) {
                if (ehTerceiro) {
                    grupoEmpresa.classList.remove('elemento-escondido');
                    inputEmpresa.required = true;
                } else {
                    grupoEmpresa.classList.add('elemento-escondido');
                    inputEmpresa.required = false;
                    inputEmpresa.value = "";
                }
            }
        }

        function atualizarMatrizNRs() {
            let funcaoSelecionada = document.getElementById('funcao').value;
            const container = document.getElementById('containerMatrizNRs');
            if (!container) return;

            if (funcaoSelecionada) {
                funcaoSelecionada = funcaoSelecionada.trim();
                if (funcaoSelecionada === "OPERADOR_INDUSTRIAL") funcaoSelecionada = "Operador Industrial";
                if (funcaoSelecionada === "MECANICO") funcaoSelecionada = "Mecânico";
                if (funcaoSelecionada === "ELETRICISTA") funcaoSelecionada = "Eletricista";
                if (funcaoSelecionada === "INSTRUMENTISTA") funcaoSelecionada = "Instrumentista";
                if (funcaoSelecionada === "PINTOR_INDUSTRIAL") funcaoSelecionada = "Pintor Industrial";
                if (funcaoSelecionada === "CALDEIREIRO") funcaoSelecionada = "Caldeireiro";
                if (funcaoSelecionada === "ENCANADOR_INDUSTRIAL") funcaoSelecionada = "Encanador Industrial";
                if (funcaoSelecionada === "SOLDADOR") funcaoSelecionada = "Soldador";
            }

            container.innerHTML = '';

            if (!funcaoSelecionada || !matrizNRsPorFuncao[funcaoSelecionada]) {
                container.innerHTML = '<p class="texto-informativo-vazio">Selecione uma função para carregar as NRs obrigatórias...</p>';
                return;
            }

            const nrsObrigatorias = matrizNRsPorFuncao[funcaoSelecionada];
            let gridHTML = '<div class="grid-nrs">';

            nrsObrigatorias.forEach(nr => {
                const dadosNR = bibliotecaRequisitos[nr];
                gridHTML += `
                    <div class="card-nr-item">
                        <div class="alinhamento-checkbox-label" style="margin-bottom: 8px;">
                            <input type="checkbox" id="check_${nr}" checked disabled class="checkbox-customizado">
                            <label class="label-customizado-aviso" style="color: #eab308;" for="check_${nr}">${nr}</label>
                        </div>
                        <small style="color: #cbd5e1; display: block; margin-bottom: 8px;">${dadosNR.nome}</small>
                        <div style="margin-bottom: 8px;">
                            <label class="rotulo-data">Realização:</label>
                            <input type="date" id="realizacao_${nr}" 
                                   onchange="calcularVencimentoDireto('realizacao_${nr}', 'validade_${nr}', ${dadosNR.validade})">
                        </div>
                        <div>
                            <label class="rotulo-data">Vencimento:</label>
                            <input type="date" id="validade_${nr}" readonly>
                        </div>
                    </div>`;
            });

            gridHTML += '</div>';
            container.innerHTML = gridHTML;
        }

        // ==========================================
        // 3. ENVIO DOS DADOS (SUBMIT FORM)
        // ==========================================
        document.getElementById('btnDispararSubmit').addEventListener('click', async () => {
            const form = document.getElementById('formColaborador');
            if (!form.checkValidity()) {
                form.reportValidity();
                return;
            }

            const funcaoSelecionada = document.getElementById('funcao').value;
            const perfilSelecionado = document.getElementById('perfil').value;
            const urlParams = new URLSearchParams(window.location.search);
            const ehEdicao = urlParams.has('id');

            const colaboradorData = {
                nome: document.getElementById('nome').value,
                matricula: document.getElementById('matricula').value,
                funcao: funcaoSelecionada,
                perfil: perfilSelecionado,
                usuario: document.getElementById('usuario').value.trim() || null,
                senha: document.getElementById('senha').value.trim() || null,
                realizacaoAso: document.getElementById('realizacaoAso').value || null,
                validadeAso: document.getElementById('validadeAso').value || null,
                realizacaoReciclagemPt: document.getElementById('realizacaoReciclagemPt').value || null,
                validadeReciclagemPt: document.getElementById('validadeReciclagemPt').value || null,
                podeSolicitarPt: (perfilSelecionado === 'OFICIAL' || perfilSelecionado === 'ENCARREGADO' || perfilSelecionado === 'ADMIN_SESMT'),
                podeEmitirPt: (perfilSelecionado === 'OPERADOR_INDUSTRIAL') ? true : (document.getElementById('podeEmitirPT') ? document.getElementById('podeEmitirPT').checked : false),
                ehTerceiro: document.getElementById('ehTerceiro').value === 'true',
                empresa: document.getElementById('ehTerceiro').value === 'true' ? document.getElementById('empresa').value.trim() : 'Interno'
            };

            Object.keys(bibliotecaRequisitos).forEach(coluna => {
                const inputValidadeCalculada = document.getElementById(`validade_${coluna}`);
                const inputRealizacaoDigitada = document.getElementById(`realizacao_${coluna}`);
                const propriedadeRealizacaoJava = "realizacao" + coluna;

                if (ehEdicao) {
                    const chaveProcuraValidade = coluna.toLowerCase();
                    const chaveProcuraRealizacao = "realizacao" + coluna.toLowerCase();

                    colaboradorData[coluna.toLowerCase()] = (inputValidadeCalculada && inputValidadeCalculada.value) ? inputValidadeCalculada.value : (dadosTreinamentosDoBanco[chaveProcuraValidade] || null);
                    colaboradorData[propriedadeRealizacaoJava] = (inputRealizacaoDigitada && inputRealizacaoDigitada.value) ? inputRealizacaoDigitada.value : (dadosTreinamentosDoBanco[chaveProcuraRealizacao] || null);
                } else {
                    const checkbox = document.getElementById(`check_${coluna}`);
                    if (checkbox && inputValidadeCalculada && inputValidadeCalculada.value) {
                        colaboradorData[coluna.toLowerCase()] = inputValidadeCalculada.value;
                        colaboradorData[propriedadeRealizacaoJava] = inputRealizacaoDigitada.value;
                    } else {
                        colaboradorData[coluna.toLowerCase()] = null;
                        colaboradorData[propriedadeRealizacaoJava] = null;
                    }
                }
            });

            let url = '/api/funcionarios';
            let metodo = 'POST';

            if (ehEdicao) {
                const idColaborador = urlParams.get('id');
                url = `/api/funcionarios/${idColaborador}`;
                metodo = 'PUT';
                colaboradorData.id = parseInt(idColaborador);
            }

            try {
                const response = await fetch(url, {
                    method: metodo,
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(colaboradorData)
                });

                if (response.ok) {
                    alert('⚡ Colaborador e histórico completo de SSMA salvos com sucesso!');
                    window.location.href = '/admin.html';
                } else {
                    const erroTexto = await response.text();
                    alert('Erro ao salvar colaborador: ' + erroTexto);
                }
            } catch (err) {
                alert('Erro de comunicação com o servidor.');
                console.error(err);
            }
        });

        // ==========================================
        // 4. CARREGAMENTO EM CASO DE EDIÇÃO (ONLOAD)
        // ==========================================
        async function verificarEdicao() {
            const urlParams = new URLSearchParams(window.location.search);
            const idColaborador = urlParams.get('id');

            if (idColaborador) {
                try {
                    const response = await fetch(`/api/funcionarios/${idColaborador}`);
                    if (!response.ok) return;

                    const f = await response.json();
                    dadosTreinamentosDoBanco = { ...f };

                    document.getElementById('nome').value = f.nome || '';
                    document.getElementById('matricula').value = f.matricula || '';
                    document.getElementById('usuario').value = f.usuario || '';
                    document.getElementById('senha').value = f.senha || '';
                    document.getElementById('perfil').value = f.perfil || '';

                    if (document.getElementById('ehTerceiro')) {
                        document.getElementById('ehTerceiro').value = f.ehTerceiro ? 'true' : 'false';
                        toggleEmpresa();
                    }
                    if (f.ehTerceiro && document.getElementById('empresa')) {
                        document.getElementById('empresa').value = f.empresa || '';
                    }

                    ajustarCamposLogin();

                    const campoEmitir = document.getElementById('podeEmitirPT');
                    if (campoEmitir && f.perfil === 'ENCARREGADO') {
                        if (f.podeEmitirPt) campoEmitir.checked = true;
                    }

                    Object.keys(f).forEach(chave => {
                        const chaveLimpa = chave.toLowerCase().replace(/_/g, '');
                        dadosTreinamentosDoBanco[chaveLimpa] = f[chave];
                    });

                    if (dadosTreinamentosDoBanco['realizacaoaso']) {
                        document.getElementById('realizacaoAso').value = String(dadosTreinamentosDoBanco['realizacaoaso']).split('T')[0];
                    }
                    if (dadosTreinamentosDoBanco['validadeaso']) {
                        document.getElementById('validadeAso').value = String(dadosTreinamentosDoBanco['validadeaso']).split('T')[0];
                    }
                    if (dadosTreinamentosDoBanco['realizacaoreciclagempt']) {
                        document.getElementById('realizacaoReciclagemPt').value = String(dadosTreinamentosDoBanco['realizacaoreciclagempt']).split('T')[0];
                    }
                    if (dadosTreinamentosDoBanco['validadereciclagempt']) {
                        document.getElementById('validadeReciclagemPt').value = String(dadosTreinamentosDoBanco['validadereciclagempt']).split('T')[0];
                    }

                    const selectFuncao = document.getElementById('funcao');
                    const funcaoBanco = f.funcao;
                    if (funcaoBanco && selectFuncao) {
                        const funcaoTratada = String(funcaoBanco).trim();
                        let encontrouOpcao = false;
                        for (let i = 0; i < selectFuncao.options.length; i++) {
                            if (selectFuncao.options[i].value.toLowerCase() === funcaoTratada.toLowerCase()) {
                                selectFuncao.selectedIndex = i;
                                encontrouOpcao = true;
                                break;
                            }
                        }
                        if (!encontrouOpcao) {
                            const novaOpcao = document.createElement('option');
                            novaOpcao.value = funcaoTratada;
                            novaOpcao.text = funcaoTratada;
                            novaOpcao.selected = true;
                            selectFuncao.add(novaOpcao);
                        }
                    }
                    
                    atualizarMatrizNRs();
                    document.querySelector('.titulo-secao').innerText = '✏️ Editar Registro Técnico de Colaborador';
                } catch (e) {
                    console.error("Erro crítico na carga de dados de edição:", e);
                }
            } else {
                ajustarCamposLogin();
            }
        }

        window.onload = verificarEdicao;
