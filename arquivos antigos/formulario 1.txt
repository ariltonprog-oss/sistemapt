            <textarea id="descricao" rows="3" required></textarea>
        </div>
        <div class="form-group">
            <label for="solicitante">Solicitante</label>
            <select id="solicitante" required>
                <option value="">Carregando solicitantes...</option>
            </select>
        </div>
        <div class="form-group">
            <label for="dataEmissao">Data Prorrogada/Emissão</label>
            <input type="date" id="dataEmissao" required>

<script>
    const API_BASE = "http://localhost:8080/api";
    const usuarioLogado = JSON.parse(localStorage.getItem('usuarioLogado')) || { empresaId: 1 };

    // Executa ao abrir a página para preencher o Select com funcionários reais da empresa
    async function carregarSolicitantes() {
        try {
            // Rota fictícia filtrando pela empresa logada
            const response = await fetch(`${API_BASE}/funcionarios?empresaId=${usuarioLogado.empresaId}`);
            const funcionarios = await response.json();
            
            const select = document.getElementById('solicitante');
            select.innerHTML = '<option value="">Selecione o Solicitante</option>';
            
            funcionarios.forEach(func => {
                if(func.perfil === 'SOLICITANTE') {
                    const opt = document.createElement('option');
                    opt.value = func.id;
                    opt.textContent = func.nome;
                    select.appendChild(opt);
                }
            });
        } catch (error) {
            console.error("Erro ao buscar solicitantes:", error);
        }
    }

    document.getElementById('ptForm').addEventListener('submit', async (e) => {
        e.preventDefault();

        const ptData = {
            descricao: document.getElementById('descricao').value,
            solicitanteId: parseInt(document.getElementById('solicitante').value),
            dataEmissao: document.getElementById('dataEmissao').value, // Formato ISO yyyy-MM-dd nativo do input
            empresaId: usuarioLogado.empresaId
        };

        try {
            const response = await fetch(`${API_BASE}/permissoes-trabalho`, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(ptData)
            });

            if (response.ok) {
                alert("Permissão de Trabalho solicitada com sucesso!");
                document.getElementById('ptForm').reset();
            } else {
                alert("Erro ao salvar a PT no servidor.");
            }
        } catch (error) {
            alert("Erro ao conectar com o Back-end.");
        }
    });

    // Inicializa os dados na tela
    carregarSolicitantes();
</script>
